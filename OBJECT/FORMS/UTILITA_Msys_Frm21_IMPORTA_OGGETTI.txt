'//#03.01# radice predefinita: ...\<mini-progetto>\MDB\db.mdb  ->  la cartella che contiene i mini-progetti
Private Function RadicePredefinita_pf() As String
    Dim p_s As String
    p_s = Application.CurrentProject.Path
    If UCase(Right(p_s, 4)) = "\MDB" Then p_s = Left(p_s, Len(p_s) - 4)
    p_s = Left(p_s, InStrRev(p_s, "\"))
    RadicePredefinita_pf = p_s
End Function

'//#03.03# combinata 1 = mini-progetti (catalogo, altrimenti cartelle della radice)
Private Sub CaricaCmb01_Sub()
    Dim rs As DAO.Recordset
    Dim lista_s As String
    Dim n As Long
    Dim cart As Variant
    Dim mdb_s As String
    Dim nomeCart_s As String
    Dim nDb As Long

    lista_s = ""
    n = 0

    If CatalogoPresente_b() Then
        Set rs = CurrentDb.OpenRecordset( _
            "SELECT L.COD_SUB_PROGETTO_s, L.Database_s, Count(*) AS N FROM " & TAB_CATALOGO & " AS L " & _
            "WHERE L.Database_s Like '" & Replace(Radice_pf(), "'", "''") & "*' " & _
            "GROUP BY L.COD_SUB_PROGETTO_s, L.Database_s ORDER BY L.COD_SUB_PROGETTO_s;", dbOpenSnapshot)
        Do While Not rs.EOF
            lista_s = lista_s & VL(rs!Database_s & "") & ";" & VL(rs!COD_SUB_PROGETTO_s & "") & ";" & _
                      VL("CATALOGO " & rs!n & " tab.") & ";"
            n = n + 1
            rs.MoveNext
        Loop
        rs.Close
        Set rs = Nothing
    End If

    If n = 0 And oFile.CartellaEsiste(Radice_pf()) Then
        For Each cart In oFile.ElencoCartelle(Radice_pf())
            nomeCart_s = Mid(cart, InStrRev(cart, "\") + 1)
            If Not (UCase(nomeCart_s) Like "AA_*" Or UCase(nomeCart_s) = "OLD" Or UCase(nomeCart_s) Like "BACKUP*") Then
                mdb_s = cart & "\MDB\"
                If oFile.CartellaEsiste(mdb_s) Then
                    nDb = ElencoDatabase_pc(mdb_s).Count
                    If nDb > 0 Then
                        lista_s = lista_s & VL(mdb_s) & ";" & VL(nomeCart_s) & ";" & VL("CARTELLA " & nDb & " db") & ";"
                        n = n + 1
                    End If
                End If
            End If
        Next cart
    End If

    Me.Cmb_01_txt.RowSource = lista_s
    Me.Cmb_01_txt.Value = Null
End Sub


'//#03.04# scelta del mini-progetto: carica i suoi database nella combinata 2
Private Sub Cmb_01_TXT_AfterUpdate()
    Dim cart_s As String
    Dim lista_s As String
    Dim elenco_s As String
    Dim f As Variant
    Dim nome_s As String

    Call Reset_Sub
    cart_s = Nz(Me.Cmb_01_txt.Value, "")
    If cart_s = "" Then Exit Sub

    Me.CODICE_PROGETTO_TXT = Nz(Me.Cmb_01_txt.Column(1), "")
    lista_s = ""
    elenco_s = ""
    For Each f In ElencoDatabase_pc(cart_s)
        nome_s = NomeFile_pf(CStr(f))
        lista_s = lista_s & VL(CStr(f)) & ";" & VL(nome_s) & ";" & VL(TipoDatabase_pf(nome_s)) & ";"
        elenco_s = elenco_s & "   " & Left(nome_s & Space(50), 50) & TipoDatabase_pf(nome_s) & vbCrLf
    Next f
    Call ImpostaCombinata_Sub(Me.Cmb_02_txt, 3, "0;5000;1500")
    Me.Cmb_02_txt.RowSource = lista_s
    Me.Cmb_02_txt.Value = Null
    Call Titolo_Sub
    Me.TXT_OBJECT_ESTERNI = "MINI-PROGETTO: " & Me.CODICE_PROGETTO_TXT & vbCrLf & "CARTELLA: " & cart_s & vbCrLf & vbCrLf & _
                            "DATABASE TROVATI:" & vbCrLf & IIf(elenco_s = "", "   (nessuno)" & vbCrLf, elenco_s) & vbCrLf & _
                            "Scegli ora nella combinata 2 il database da cui importare."
End Sub

Private Sub ImpostaSorgente_Sub(ByVal path_s As String)
    Me.sourceDBPath_s_Txt = path_s
    If path_s = "" Then
        Call AbilitaComandi_Sub(False)
        Exit Sub
    End If
    If StrComp(path_s, CurrentProject.FullName, vbTextCompare) = 0 Then
        MsgBox "Il database sorgente e' il database corrente: scegline un altro.", vbExclamation, Me.Name
        Me.sourceDBPath_s_Txt = ""
        Call AbilitaComandi_Sub(False)
        Exit Sub
    End If
    Call AbilitaComandi_Sub(True)
    Call Titolo_Sub
    Me.TXT_OBJECT_ESTERNI = Riepilogo_pf(path_s, False)
End Sub


Private Sub Reset_Sub()
    Me.Cmb_02_txt.RowSource = ""
    Me.Cmb_02_txt.Value = Null
    Me.sourceDBPath_s_Txt = ""
    Call AbilitaComandi_Sub(False)
End Sub

'//=====================================================================================================================//
'//#05# MOTORE: CANCELLA o IMPORTA una categoria (o tutte) del database sorgente
'//  par_Azione_s = "CANCELLA" | "IMPORTA"
'//  par_Tipo_s   = "TUTTI" | "TABLE" | "QUERY" | "FORMS" | "REPORTS" | "MACRO" | "MODULI"
'//  par_Silenzioso_b = True: nessun messaggio (uso da codice). Restituisce il testo del log.
'//=====================================================================================================================//
Public Function Esegui_pf(ByVal par_Azione_s As String, ByVal par_Tipo_s As String, _
                          Optional ByVal par_Sorgente_s As String = "", Optional ByVal par_Silenzioso_b As Boolean = False) As String
    Dim src_s As String
    Dim tipi_v As Variant
    Dim t As Variant
    Dim v As Variant
    Dim elenchi As Object
    Dim col As Collection
    Dim nTot As Long, nOk As Long, nNo As Long, nReg As Long
    Dim log_s As String
    Dim esito_s As String
    Dim reg_s As String
    Dim fatto_b As Boolean
    Dim conferma_s As String
    Dim pos_s As String

    On Error GoTo Esegui_Err
    pos_s = "05.01"
    src_s = par_Sorgente_s
    If src_s = "" Then src_s = Nz(Me.sourceDBPath_s_Txt, "")

    '//#05.01# controlli
    If src_s = "" Then Esegui_pf = "SCEGLI PRIMA IL DATABASE SORGENTE.": GoTo Esegui_Msg
    If Not oFile.FileEsiste(src_s) Then Esegui_pf = "IL DATABASE SORGENTE NON ESISTE: " & src_s: GoTo Esegui_Msg
    If StrComp(src_s, CurrentProject.FullName, vbTextCompare) = 0 Then Esegui_pf = "IL SORGENTE E' IL DATABASE CORRENTE.": GoTo Esegui_Msg

    If par_Tipo_s = "TUTTI" Then
        If par_Azione_s = "CANCELLA" Then
            tipi_v = Array("FORMS", "REPORTS", "MACRO", "QUERY", "MODULI", "TABLE")
        Else
            tipi_v = Array("TABLE", "QUERY", "FORMS", "REPORTS", "MACRO", "MODULI")
        End If
    Else
        tipi_v = Array(par_Tipo_s)
    End If

    '//#05.02# elenco degli oggetti del sorgente, per categoria
    pos_s = "05.02"
    DoCmd.Hourglass True
    Set elenchi = CreateObject("Scripting.Dictionary")
    conferma_s = ""
    For Each t In tipi_v
        Set col = ElencoSorgente_pc(CStr(t), src_s)
        elenchi.Add CStr(t), col
        nTot = nTot + col.Count
        conferma_s = conferma_s & "   " & NomeTipo_pf(CStr(t)) & ": " & col.Count & vbCrLf
    Next t
    DoCmd.Hourglass False
    If nTot = 0 Then Esegui_pf = "NEL DATABASE SORGENTE NON CI SONO OGGETTI DI QUESTO TIPO.": GoTo Esegui_Msg

    If Not par_Silenzioso_b And Nz(Me.ctr_AVVISO_ABILITATO_TXT, True) Then
        conferma_s = par_Azione_s & " " & IIf(par_Tipo_s = "TUTTI", "TUTTE LE CATEGORIE", NomeTipo_pf(par_Tipo_s)) & vbCrLf & _
                     "SORGENTE: " & src_s & vbCrLf & "CORRENTE: " & CurrentProject.FullName & vbCrLf & vbCrLf & conferma_s & vbCrLf & _
                     IIf(par_Azione_s = "CANCELLA", "Nel database corrente verranno CANCELLATI gli oggetti con questi nomi.", _
                         "Gli oggetti gia' presenti con lo stesso nome verranno SOSTITUITI e registrati in " & TAB_OGGETTI & ".") & _
                     vbCrLf & vbCrLf & "Continuo?"
        If MsgBox(conferma_s, vbYesNo + vbQuestion + vbDefaultButton2, Me.Name) <> vbYes Then
            Esegui_pf = "OPERAZIONE ANNULLATA."
            Exit Function
        End If
    End If

    '//#05.03# esecuzione
    pos_s = "05.03"
    DoCmd.Hourglass True
    log_s = "TB99 " & par_Azione_s & " " & par_Tipo_s & " - " & Format(Now, "dd/mm/yyyy hh:nn:ss") & vbCrLf & _
            "SORGENTE: " & src_s & vbCrLf & "CORRENTE: " & CurrentProject.FullName & vbCrLf & String(110, "=") & vbCrLf
    If par_Azione_s = "IMPORTA" Then Call PreparaAnalisi_Sub(src_s)

    For Each t In tipi_v
        Set col = elenchi(CStr(t))
        log_s = log_s & vbCrLf & NomeTipo_pf(CStr(t)) & " (" & col.Count & ")" & vbCrLf
        For Each v In col
            SysCmd acSysCmdSetStatus, par_Azione_s & " " & NomeTipo_pf(CStr(t)) & ": " & v
            reg_s = ""
            If par_Azione_s = "CANCELLA" Then
                esito_s = Cancella_pf(CStr(t), CStr(v), fatto_b)
            ElseIf CStr(t) = "TABLE" Then
                esito_s = Collega_pf(src_s, CStr(v), fatto_b)
            Else
                esito_s = Importa_pf(src_s, CStr(t), CStr(v), fatto_b)
                If fatto_b Then
                    reg_s = Registra_pf(src_s, CStr(t), CStr(v))
                    nReg = nReg + 1
                End If
            End If
            If fatto_b Then nOk = nOk + 1 Else nNo = nNo + 1
            log_s = log_s & "   " & Left(v & Space(60), 60) & " " & esito_s & IIf(reg_s <> "", "  >> " & reg_s, "") & vbCrLf
        Next v
    Next t

    log_s = log_s & vbCrLf & String(110, "=") & vbCrLf & "ESEGUITI: " & nOk & "   NON ESEGUITI: " & nNo & _
            IIf(par_Azione_s = "IMPORTA", "   REGISTRATI IN " & TAB_OGGETTI & ": " & nReg, "") & vbCrLf
    log_s = log_s & "LOG: " & ScriviFileLog_pf(log_s) & vbCrLf

    SysCmd acSysCmdClearStatus
    DoCmd.Hourglass False
    Application.RefreshDatabaseWindow
    On Error Resume Next
    oLog.ScriviLogEvento "Esegui_pf", par_Azione_s & " " & par_Tipo_s & " da " & src_s & ": eseguiti " & nOk & ", non eseguiti " & nNo
    Me.TXT_OBJECT_ESTERNI = log_s
    On Error GoTo 0
    If Not par_Silenzioso_b Then
        MsgBox "OPERAZIONE TERMINATA" & vbCrLf & vbCrLf & "Eseguiti: " & nOk & vbCrLf & "Non eseguiti: " & nNo & _
               IIf(par_Azione_s = "IMPORTA", vbCrLf & "Registrati in " & TAB_OGGETTI & ": " & nReg, ""), _
               IIf(nNo = 0, vbInformation, vbExclamation), Me.Name
    End If
    Esegui_pf = log_s
    Exit Function

Esegui_Msg:
    DoCmd.Hourglass False
    If Not par_Silenzioso_b Then MsgBox Esegui_pf, vbExclamation, Me.Name
    Exit Function

Esegui_Err:
    esito_s = "ERRORE " & Err.Number & " (pos. " & pos_s & "): " & Err.Description
    On Error Resume Next
    SysCmd acSysCmdClearStatus
    DoCmd.Hourglass False
    oLog.ScriviErrore "Esegui_pf", pos_s, Err.Number, esito_s
    Esegui_pf = log_s & vbCrLf & esito_s
    Me.TXT_OBJECT_ESTERNI = Esegui_pf
    If Not par_Silenzioso_b Then MsgBox esito_s, vbCritical, Me.Name
End Function


'//#05.04# oggetti del sorgente di una categoria (senza quelli protetti)
Private Function ElencoSorgente_pc(ByVal par_Tipo_s As String, ByVal par_Src_s As String) As Collection
    Dim col As New Collection
    Dim v As Variant
    Dim nome_s As String
    For Each v In oOgg.ElencoOggetti(par_Tipo_s, "*", par_Src_s, True)
        nome_s = Mid(v, InStr(v, "|") + 1)
        If Not Protetto_b(nome_s) Then col.Add nome_s
    Next v
    Set ElencoSorgente_pc = col
End Function


'//#05.05# cancella un oggetto nel database corrente
Private Function Cancella_pf(ByVal par_Tipo_s As String, ByVal par_Nome_s As String, ByRef par_Fatto_b As Boolean) As String
    On Error GoTo Cancella_Err
    par_Fatto_b = False
    If Not oOgg.OggettoEsiste(par_Nome_s, par_Tipo_s) Then Cancella_pf = "NON PRESENTE": Exit Function
    If SysCmd(acSysCmdGetObjectState, oOgg.TipoAccess(par_Tipo_s), par_Nome_s) <> 0 Then Cancella_pf = "APERTO: NON TOCCATO": Exit Function
    If par_Tipo_s = "TABLE" Then
        If CurrentDb.TableDefs(par_Nome_s).Connect = "" Then Cancella_pf = "TABELLA LOCALE: NON TOCCATA": Exit Function
    End If
    DoCmd.DeleteObject oOgg.TipoAccess(par_Tipo_s), par_Nome_s
    par_Fatto_b = True
    Cancella_pf = IIf(par_Tipo_s = "TABLE", "COLLEGAMENTO CANCELLATO", "CANCELLATO")
    Exit Function
Cancella_Err:
    Cancella_pf = "ERRORE " & Err.Number & ": " & Err.Description
End Function


'//#05.06# importa un oggetto (sostituisce quello con lo stesso nome, se non e' aperto)
Private Function Importa_pf(ByVal par_Src_s As String, ByVal par_Tipo_s As String, ByVal par_Nome_s As String, ByRef par_Fatto_b As Boolean) As String
    Dim c_b As Boolean
    On Error GoTo Importa_Err
    par_Fatto_b = False
    c_b = oOgg.OggettoEsiste(par_Nome_s, par_Tipo_s)
    If c_b Then
        If SysCmd(acSysCmdGetObjectState, oOgg.TipoAccess(par_Tipo_s), par_Nome_s) <> 0 Then Importa_pf = "APERTO: NON SOSTITUITO": Exit Function
    End If
    If oOgg.ImportaOggetto(par_Src_s, par_Tipo_s, par_Nome_s, vbNullString, True) Then
        par_Fatto_b = True
        Importa_pf = IIf(c_b, "SOSTITUITO", "IMPORTATO")
    Else
        Importa_pf = "NON IMPORTATO " & oOgg.UltimoErrore
    End If
    Exit Function
Importa_Err:
    Importa_pf = "ERRORE " & Err.Number & ": " & Err.Description
End Function


'//#05.07# collega una tabella del sorgente (se nel sorgente e' un link, punto alla tabella fisica)
Private Function Collega_pf(ByVal par_Src_s As String, ByVal par_Nome_s As String, ByRef par_Fatto_b As Boolean) As String
    Dim dbS As DAO.Database
    Dim dbC As DAO.Database
    Dim tdS As DAO.TableDef
    Dim tdN As DAO.TableDef
    Dim connect_s As String
    Dim sorg_s As String
    Dim c_b As Boolean

    On Error GoTo Collega_Err
    par_Fatto_b = False
    Set dbC = CurrentDb
    c_b = oOgg.OggettoEsiste(par_Nome_s, "TABLE")
    If c_b Then
        If dbC.TableDefs(par_Nome_s).Connect = "" Then Collega_pf = "TABELLA LOCALE: NON TOCCATA": GoTo Collega_Exit
        If SysCmd(acSysCmdGetObjectState, acTable, par_Nome_s) <> 0 Then Collega_pf = "APERTA: NON TOCCATA": GoTo Collega_Exit
    End If

    Set dbS = DBEngine.Workspaces(0).OpenDatabase(par_Src_s, False, True)
    Set tdS = dbS.TableDefs(par_Nome_s)
    If tdS.Connect <> "" Then
        connect_s = tdS.Connect
        sorg_s = tdS.SourceTableName
    Else
        connect_s = ";DATABASE=" & par_Src_s
        sorg_s = par_Nome_s
    End If
    Set tdS = Nothing
    dbS.Close
    Set dbS = Nothing

    If c_b Then dbC.TableDefs.Delete par_Nome_s
    Set tdN = dbC.CreateTableDef(par_Nome_s)
    tdN.Connect = connect_s
    tdN.SourceTableName = sorg_s
    dbC.TableDefs.Append tdN
    par_Fatto_b = True
    Collega_pf = IIf(c_b, "RICOLLEGATA ", "COLLEGATA ") & Mid(connect_s, 11)

Collega_Exit:
    Set tdN = Nothing
    Set dbC = Nothing
    Exit Function
Collega_Err:
    Collega_pf = "ERRORE " & Err.Number & ": " & Err.Description
    On Error Resume Next
    If Not dbS Is Nothing Then dbS.Close
    Resume Collega_Exit
End Function

'//#06.01# prepara: tabelle del catalogo che stanno nella cartella del sorgente, query del sorgente, cache
Private Sub PreparaAnalisi_Sub(ByVal par_Src_s As String)
    Dim rs As DAO.Recordset
    Dim cart_s As String

    Set m_TabMini = CreateObject("Scripting.Dictionary")
    Set m_Cache = CreateObject("Scripting.Dictionary")
    Call CaricaQuery_Sub(par_Src_s)
    If Not OggettiPresente_b() Then Exit Sub

    cart_s = Left(par_Src_s, InStrRev(par_Src_s, "\"))
    Set rs = CurrentDb.OpenRecordset("SELECT Name FROM " & TAB_CATALOGO & " WHERE Database_s = '" & Replace(cart_s, "'", "''") & "';", dbOpenSnapshot)
    Do While Not rs.EOF
        If Not m_TabMini.Exists(LCase(rs!Name)) Then m_TabMini.Add LCase(rs!Name), CStr(rs!Name)
        rs.MoveNext
    Loop
    rs.Close
    Set rs = Nothing
End Sub


'//#06.02# query del database SORGENTE con il loro SQL (le dipendenze si seguono anche se non sono state importate)
Private Sub CaricaQuery_Sub(ByVal par_Src_s As String)
    Dim dbS As DAO.Database
    Dim qd As DAO.QueryDef
    Set m_Query = New Collection
    Set m_SqlSrc = CreateObject("Scripting.Dictionary")
    On Error GoTo Esci
    Set dbS = DBEngine.Workspaces(0).OpenDatabase(par_Src_s, False, True)
    For Each qd In dbS.QueryDefs
        If Left(qd.Name, 1) <> "~" Then
            m_Query.Add qd.Name
            m_SqlSrc(LCase(qd.Name)) = qd.sql
        End If
    Next qd
Esci:
    On Error Resume Next
    If Not dbS Is Nothing Then dbS.Close
    Set dbS = Nothing
End Sub


'//#06.03# registra l'oggetto importato: una riga per ogni tabella del mini-progetto che usa
Private Function Registra_pf(ByVal par_Src_s As String, ByVal par_Tipo_s As String, ByVal par_Nome_s As String) As String
    Dim trovate_s As String
    Dim v As Variant
    Dim file_s As String
    Dim cart_s As String
    Dim tipoReg_s As String
    Dim elenco_s As String
    Dim db As DAO.Database

    On Error GoTo Registra_Err
    If Not OggettiPresente_b() Then Registra_pf = "catalogo assente: non registrato": Exit Function

    file_s = Mid(par_Src_s, InStrRev(par_Src_s, "\") + 1)
    cart_s = Left(par_Src_s, InStrRev(par_Src_s, "\"))
    tipoReg_s = IIf(par_Tipo_s = "MODULI", "MODULES", par_Tipo_s)

    trovate_s = TabelleUsate_pf(par_Tipo_s, par_Nome_s, 0)

    Set db = CurrentDb
    db.Execute "DELETE FROM " & TAB_OGGETTI & " WHERE NOME_OGGETTO_s = " & Q(par_Nome_s) & " AND TIPO_OGGETTO_s = " & Q(tipoReg_s) & _
               " AND [Database] = " & Q(file_s) & " AND Database_s = " & Q(cart_s) & ";", dbFailOnError

    elenco_s = ""
    For Each v In Split(trovate_s, "|")
        If v <> "" Then
            db.Execute "INSERT INTO " & TAB_OGGETTI & " (Name, NOME_OGGETTO_s, TIPO_OGGETTO_s, [Database], Database_s, Scel_b, DATAINS_d) VALUES (" & _
                       Q(m_TabMini(v)) & ", " & Q(par_Nome_s) & ", " & Q(tipoReg_s) & ", " & Q(file_s) & ", " & Q(cart_s) & ", False, Now());", dbFailOnError
            elenco_s = elenco_s & IIf(elenco_s = "", "", ", ") & m_TabMini(v)
        End If
    Next v
    If elenco_s = "" Then
        db.Execute "INSERT INTO " & TAB_OGGETTI & " (Name, NOME_OGGETTO_s, TIPO_OGGETTO_s, [Database], Database_s, Scel_b, DATAINS_d) VALUES (" & _
                   "Null, " & Q(par_Nome_s) & ", " & Q(tipoReg_s) & ", " & Q(file_s) & ", " & Q(cart_s) & ", False, Now());", dbFailOnError
        Registra_pf = "nessuna tabella del mini-progetto"
    Else
        Registra_pf = "tabelle: " & elenco_s
    End If
    Set db = Nothing
    Exit Function
Registra_Err:
    Registra_pf = "REGISTRAZIONE: ERRORE " & Err.Number & " " & Err.Description
End Function


'//#06.04# tabelle del mini-progetto usate dall'oggetto (direttamente o tramite query e sottomaschere)
'//         restituisce "|chiave1|chiave2|" con le chiavi lcase di m_TabMini
Private Function TabelleUsate_pf(ByVal par_Tipo_s As String, ByVal par_Nome_s As String, ByVal par_Liv As Integer) As String
    Dim k_s As String
    Dim testo_s As String
    Dim ris_s As String
    Dim v As Variant
    Dim Q As Variant
    Dim sub_s As String
    Dim p As Long, p2 As Long

    k_s = LCase(par_Tipo_s & "|" & par_Nome_s)
    If m_Cache.Exists(k_s) Then TabelleUsate_pf = m_Cache(k_s): Exit Function
    m_Cache(k_s) = "|"                                   '//evita i cicli
    If par_Liv > PROFONDITA_MAX Then TabelleUsate_pf = "|": Exit Function

    testo_s = TestoOggetto_pf(par_Tipo_s, par_Nome_s)
    ris_s = "|"
    If testo_s <> "" Then
        '//tabelle dirette
        For Each v In m_TabMini.Keys
            If ContieneNome_b(testo_s, m_TabMini(v)) Then ris_s = ris_s & v & "|"
        Next v
        '//query usate
        For Each Q In m_Query
            If StrComp(Q, par_Nome_s, vbTextCompare) <> 0 Or par_Tipo_s <> "QUERY" Then
                If ContieneNome_b(testo_s, CStr(Q)) Then ris_s = Unisci_pf(ris_s, TabelleUsate_pf("QUERY", CStr(Q), par_Liv + 1))
            End If
        Next Q
        '//sottomaschere e sottoreport
        If par_Tipo_s = "FORMS" Or par_Tipo_s = "REPORTS" Then
            p = InStr(1, testo_s, "SourceObject =""", vbTextCompare)
            Do While p > 0
                p = p + Len("SourceObject =""")
                p2 = InStr(p, testo_s, """")
                If p2 = 0 Then Exit Do
                sub_s = Mid(testo_s, p, p2 - p)
                If LCase(Left(sub_s, 5)) = "form." Then
                    sub_s = Mid(sub_s, 6)
                    If oOgg.OggettoEsiste(sub_s, "FORMS") Then ris_s = Unisci_pf(ris_s, TabelleUsate_pf("FORMS", sub_s, par_Liv + 1))
                ElseIf LCase(Left(sub_s, 7)) = "report." Then
                    sub_s = Mid(sub_s, 8)
                    If oOgg.OggettoEsiste(sub_s, "REPORTS") Then ris_s = Unisci_pf(ris_s, TabelleUsate_pf("REPORTS", sub_s, par_Liv + 1))
                End If
                p = InStr(p2, testo_s, "SourceObject =""", vbTextCompare)
            Loop
        End If
    End If
    m_Cache(k_s) = ris_s
    TabelleUsate_pf = ris_s
End Function


'//#06.05# testo dell'oggetto nel database corrente: SQL della query, oppure testo esportato (form, report, macro, moduli)
Private Function TestoOggetto_pf(ByVal par_Tipo_s As String, ByVal par_Nome_s As String) As String
    Dim f_s As String
    On Error GoTo Testo_Err
    If par_Tipo_s = "QUERY" Then
        If m_SqlSrc.Exists(LCase(par_Nome_s)) Then
            TestoOggetto_pf = m_SqlSrc(LCase(par_Nome_s))
        Else
            TestoOggetto_pf = CurrentDb.QueryDefs(par_Nome_s).sql
        End If
        Exit Function
    End If
    f_s = Environ("TEMP") & "\TB99_ANALISI_" & Format(Now, "hhnnss") & "_" & Int(Rnd * 100000) & ".txt"
    Application.SaveAsText oOgg.TipoAccess(par_Tipo_s), par_Nome_s, f_s
    TestoOggetto_pf = LeggiFile_pf(f_s)
    Kill f_s
    Exit Function
Testo_Err:
    TestoOggetto_pf = ""
    On Error Resume Next
    If f_s <> "" Then Kill f_s
End Function

Private Function Riepilogo_pf(ByVal par_Src_s As String, ByVal par_Dettaglio_b As Boolean) As String
    Dim t As Variant
    Dim col As Collection
    Dim v As Variant
    Dim s As String
    Dim nCorr As Long
    Dim nReg As Long
    Dim file_s As String
    Dim cart_s As String

    On Error Resume Next
    DoCmd.Hourglass True
    file_s = Mid(par_Src_s, InStrRev(par_Src_s, "\") + 1)
    cart_s = Left(par_Src_s, InStrRev(par_Src_s, "\"))
    s = "MINI-PROGETTO: " & Nz(Me.CODICE_PROGETTO_TXT, "") & vbCrLf & _
        "SORGENTE: " & par_Src_s & vbCrLf & "CORRENTE: " & CurrentProject.FullName & vbCrLf & String(110, "-") & vbCrLf & _
        Left("CATEGORIA" & Space(22), 22) & "NEL SORGENTE   GIA' NEL CORRENTE   REGISTRATI IN " & TAB_OGGETTI & vbCrLf

Private Function Protetto_b(ByVal par_Nome_s As String) As Boolean
    Dim n_s As String
    Dim padre_s As String
    n_s = LCase(Trim(par_Nome_s))
    Protetto_b = True
    If n_s = "" Then Exit Function
    If n_s = LCase(Me.Name) Then Exit Function
    If n_s Like "utilita_msys_frm21_*" Then Exit Function
    If n_s Like "msys_tb99_*" Then Exit Function
    If n_s Like "msys_tb12_*" Then Exit Function
    If Left(n_s, 1) = "~" Then Exit Function
    On Error Resume Next
    padre_s = LCase(Me.Parent.Name)
    On Error GoTo 0
    If padre_s <> "" And n_s = padre_s Then Exit Function
    Protetto_b = False
End Function

'//i database di una cartella (mdb e accdb, senza le copie _Backup), percorsi puliti
Private Function ElencoDatabase_pc(ByVal par_Cart_s As String) As Collection
    Dim col As New Collection
    Dim v As Variant
    Dim est As Variant
    Dim p_s As String
    On Error Resume Next
    For Each est In Array("*.mdb", "*.accdb")
        For Each v In oFile.ElencoFile(par_Cart_s, CStr(est))
            p_s = Pulisci_pf(CStr(v))
            If p_s <> "" And Not (UCase(p_s) Like "*_BACKUP*") Then col.Add p_s
        Next v
    Next est
    Set ElencoDatabase_pc = col
End Function

'//toglie caratteri nulli, spazi e il prefisso \\?\ dei percorsi lunghi
Private Function Pulisci_pf(ByVal par_s As String) As String
    Dim p As Long
    p = InStr(par_s, Chr$(0))
    If p > 0 Then par_s = Left$(par_s, p - 1)
    par_s = Trim$(par_s)
    If Left$(par_s, 8) = "\\?\UNC\" Then par_s = "\\" & Mid$(par_s, 9)
    If Left$(par_s, 4) = "\\?\" Then par_s = Mid$(par_s, 5)
    Pulisci_pf = par_s
End Function

'//nome del file da un percorso (con la classe; se non risponde, dal testo)

    Dim n_s As String

    If n_s = "" Then n_s = Mid$(par_Path_s, InStrRev(par_Path_s, "\") + 1)

End Function

'//il nome compare nel testo come parola intera (prima e dopo non c'e' lettera, cifra o _)
Private Function ContieneNome_b(ByVal par_Testo_s As String, ByVal par_Nome_s As String) As Boolean
    Dim p As Long
    Dim cPrima As String
    Dim cDopo As String
    If par_Nome_s = "" Then Exit Function
    p = InStr(1, par_Testo_s, par_Nome_s, vbTextCompare)
    Do While p > 0
        If p = 1 Then cPrima = " " Else cPrima = Mid(par_Testo_s, p - 1, 1)
        cDopo = Mid(par_Testo_s, p + Len(par_Nome_s), 1)
        If Not (cPrima Like "[A-Za-z0-9_]") And Not (cDopo Like "[A-Za-z0-9_]") Then
            ContieneNome_b = True
            Exit Function
        End If
        p = InStr(p + 1, par_Testo_s, par_Nome_s, vbTextCompare)
    Loop
End Function

Private Function Unisci_pf(ByVal par_A_s As String, ByVal par_B_s As String) As String
    Dim v As Variant
    Unisci_pf = par_A_s
    For Each v In Split(par_B_s, "|")
        If v <> "" Then
            If InStr(1, Unisci_pf, "|" & v & "|", vbTextCompare) = 0 Then Unisci_pf = Unisci_pf & v & "|"
        End If
    Next v
End Function

'//legge un file di testo: UTF-16 (con BOM) oppure ANSI
Private Function LeggiFile_pf(ByVal par_File_s As String) As String
    Dim n As Integer
    Dim b() As Byte
    Dim L As Long
    n = FreeFile
    Open par_File_s For Binary Access Read As #n
    L = LOF(n)
    If L = 0 Then Close #n: Exit Function
    ReDim b(0 To L - 1)
    Get #n, , b
    Close #n
    If L >= 2 Then
        If b(0) = &HFF And b(1) = &HFE Then
            LeggiFile_pf = Mid(CStr(b), 2)
            Exit Function
        End If
    End If
    LeggiFile_pf = StrConv(b, vbUnicode)
End Function

Public Function FormEsisteNelDatabase(NomeForm_s As String) As Boolean

Private Sub AumentaDimensioni(par_NameForm_s As String)
    Dim frm As Form

    If par_NameForm_s > "" Then
    
        ' Chiudi la form se è aperta
        'If SysCmd(acSysCmdGetObjectState, acForm, "AA_PROVA") <> 0 Then
        '    DoCmd.Close acForm, "AA_PROVA", acSaveYes

Private Sub ModificaFormPrincipale(frm As Form, Larghezza As Double)
    frm.Width = CmToTwips(Larghezza)
    'DoCmd.Save
    DoEvents
End Sub


Private Sub ModificaCorpoMaschera(frm As Form, Altezza As Double)
    frm.Section(acDetail).Height = CmToTwips(Altezza)
    DoCmd.Save
    DoEvents
End Sub


Private Sub ModificaTabControl(frm As Form, Larghezza As Double, Altezza As Double)
    Dim tabCtl As Control
    On Error Resume Next
    Set tabCtl = frm("TabCtl1")
    On Error GoTo 0
    If Not tabCtl Is Nothing Then
        tabCtl.Width = CmToTwips(Larghezza)
        tabCtl.Height = CmToTwips(Altezza)
        DoCmd.Save
        DoEvents
        DoCmd.Save
    End If
      
    Set tabCtl = Nothing
End Sub


Private Sub ModificaPagineTab(frm As Form, Larghezza As Double, Altezza As Double)
    Dim tabCtl As Control
    Dim i As Integer
    
    ' Ottiene il controllo TabCtl1
    On Error Resume Next
    Set tabCtl = frm("TabCtl1")
    On Error GoTo 0
    
    ' Se il Tab Control esiste, modifica le pagine
    If Not tabCtl Is Nothing Then
        For i = 0 To tabCtl.Pages.Count - 1
            tabCtl.Pages(i).Width = CmToTwips(Larghezza)
            tabCtl.Pages(i).Height = CmToTwips(Altezza)
            DoEvents
            DoCmd.Save
        Next i
    End If
    
    ' Pulizia memoria
    Set tabCtl = Nothing
End Sub


Private Sub ModificaSottoform(frm As Form, Larghezza As Double, Altezza As Double)
    Const MIN_LARGHEZZA As Double = 5    ' Larghezza minima in cm
    Const MAX_LARGHEZZA As Double = 30   ' Larghezza massima in cm
    Const MIN_ALTEZZA As Double = 5      ' Altezza minima in cm
    Const MAX_ALTEZZA As Double = 25     ' Altezza massima in cm
    
    Dim i As Integer, ctl As Control
    Dim larghezzaEffettiva As Double, altezzaEffettiva As Double
    
    ' Verifica e limita le dimensioni nei range definiti
    larghezzaEffettiva = IIf(Larghezza < MIN_LARGHEZZA, MIN_LARGHEZZA, IIf(Larghezza > MAX_LARGHEZZA, MAX_LARGHEZZA, Larghezza))
    altezzaEffettiva = IIf(Altezza < MIN_ALTEZZA, MIN_ALTEZZA, IIf(Altezza > MAX_ALTEZZA, MAX_ALTEZZA, Altezza))
    
    For i = 1 To 12
        On Error Resume Next
        Set ctl = frm("sottoform_" & Format(i, "00"))
        On Error GoTo 0
        If Not ctl Is Nothing Then
            ' Modifica dimensioni con i valori limitati
            ctl.Width = CmToTwips(larghezzaEffettiva)
            ctl.Height = CmToTwips(altezzaEffettiva)
            Debug.Print "Modificato: "; ctl.Name; " - Larghezza: "; larghezzaEffettiva; " cm, Altezza: "; altezzaEffettiva; " cm"
            
            ' Imposta l'aspetto piatto, il colore e lo spessore del bordo
            ctl.BorderStyle = 1 ' Stile continuo (fmBorderStyleSingle)
            ctl.SpecialEffect = 0 ' Aspetto piatto (fmSpecialEffectFlat)
            ctl.BorderColor = RGB(&HBA, &H14, &H19) ' Colore bordo: #BA1419
            ctl.BorderWidth = 2 ' Spessore bordo: 2 pt
            
            DoCmd.Save
            DoEvents
        End If
        Set ctl = Nothing
    Next i
End Sub


Function CmToTwips(cm As Variant) As Long
    Const TWIPS_PER_CM As Double = 567 ' 1 cm = 567 twips
    Const MAX_TWIPS As Long = 32767 ' Limite massimo per larghezza/altezza degli oggetti
    Const MIN_TWIPS As Long = 1 ' Limite minimo per evitare valori negativi o nulli
    
    Dim result As Long
    
    On Error GoTo ErrHandler ' Abilita la gestione degli errori

    ' Controllo se il valore è numerico
    If IsNull(cm) Or Not IsNumeric(cm) Then
        MsgBox "Errore: Il valore inserito non è numerico o è nullo.", vbCritical, "Errore di conversione"
        Exit Function
    End If

    ' Controllo valori negativi
    If cm < 0 Then
        MsgBox "Errore: Il valore in cm non può essere negativo.", vbCritical, "Errore di conversione"
        Exit Function
    End If

    ' Conversione
    result = cm * TWIPS_PER_CM

    ' Applicare i limiti con messaggi di avviso
    If result > MAX_TWIPS Then
        MsgBox "Attenzione: Il valore convertito supera il limite massimo e verrà impostato a " & MAX_TWIPS & " twips.", vbInformation, "Limite superato"
        result = MAX_TWIPS
    ElseIf result < MIN_TWIPS Then
        MsgBox "Attenzione: Il valore convertito è troppo basso e verrà impostato a " & MIN_TWIPS & " twips.", vbInformation, "Limite minimo"
        result = MIN_TWIPS
    End If
    
    CmToTwips = result
    
ExitFunction:
    Exit Function

ErrHandler:
    MsgBox "Si è verificato un errore: " & Err.Description, vbExclamation, "Errore " & Err.Number

    ' //@FILE@LOG@ERRORI
    ' //.................................................................//
    Dim NomeRoutine_s As String
    Dim NumeroErrore_lng As Long
    Dim MessaggioErrore_s As String

    NomeRoutine_s = "ROUTINE: CmToTwips"
    NumeroErrore_lng = Err.Number
    MessaggioErrore_s = "ERRORE DI ESECUZIONE ROUTINE : " & Err.Description

    ScriviLogErrore NomeRoutine_s, _
                    NumeroErrore_lng, _
                    MessaggioErrore_s

    ' //.................................................................//

    Resume ExitFunction
End Function

        Me!CurrentProject_txt.BackColor = lngBianco

      '//.................................................................//

          'Dim NumeroErrore_lng As Long
          'Dim MessaggioErrore_s  As String
          
          'NomeRoutine_s = "ROUTINE: Cmb_01_FORM_TXT_GotFocus"

    Dim fileNum As Integer
    Dim TestoLog_s As String

    fileLog_s = ""

    FileLog = CurrentProject.Path & fileLog_s

'//   #10.01# Form_Resize       #10.02# MisureFrm21_Sub (misure di progetto, lette la prima volta)
'//   #10.03# AdattaFinestra    ordine: 1) maschera e tab al massimo  2) controlli interni  3) misura finale
'//=====================================================================================================================//

Private Sub Form_Resize()
    '//#10.01#
    AdattaFinestra
End Sub

Private Sub MisureFrm21_Sub()
    '//#10.02# misure di progetto della maschera e dei controlli che crescono
    Dim v As Variant
    On Error GoTo ERRORE
    Set m_F21_W = CreateObject("Scripting.Dictionary")
    Set m_F21_H = CreateObject("Scripting.Dictionary")
    m_F21_W0 = Me.Width
    m_F21_H0 = Me.Section(acDetail).Height
    m_F21_Hdr = Me.Section(acHeader).Height
    For Each v In Split("TabCtl_01|" & F21_SOLO_LARGHI & "|" & F21_LARGHI_ALTI, "|")
        m_F21_W(v) = Me.Controls(v).Width
        m_F21_H(v) = Me.Controls(v).Height
    Next v
    m_F21_Pronto = True
    Exit Sub
ERRORE:
    m_F21_Pronto = False
End Sub

Public Sub AdattaFinestra()
    '//#10.03# allarga la form allo spazio disponibile
    Dim dW As Long, dH As Long, v As Variant, t As Control
    If Not m_F21_Pronto Then MisureFrm21_Sub
    If Not m_F21_Pronto Then Exit Sub
    On Error GoTo ERRORE
    dW = Me.InsideWidth - m_F21_W0
    dH = Me.InsideHeight - m_F21_Hdr - m_F21_H0
    If dW < 0 Then dW = 0
    If dH < 0 Then dH = 0
    Set t = Me!TabCtl_01
    If t.Width = m_F21_W("TabCtl_01") + dW And t.Height = m_F21_H("TabCtl_01") + dH Then Exit Sub
    Me.Painting = False
    '// 1) maschera e tab al massimo tra misura attuale e nuova
    If Me.Width < m_F21_W0 + dW Then Me.Width = m_F21_W0 + dW
    If Me.Section(acDetail).Height < m_F21_H0 + dH Then Me.Section(acDetail).Height = m_F21_H0 + dH
    If t.Width < m_F21_W("TabCtl_01") + dW Then t.Width = m_F21_W("TabCtl_01") + dW
    If t.Height < m_F21_H("TabCtl_01") + dH Then t.Height = m_F21_H("TabCtl_01") + dH
    '// 2) controlli interni
    For Each v In Split(F21_SOLO_LARGHI, "|")
        Me.Controls(v).Width = m_F21_W(v) + dW
    Next v
    For Each v In Split(F21_LARGHI_ALTI, "|")
        Me.Controls(v).Width = m_F21_W(v) + dW
        Me.Controls(v).Height = m_F21_H(v) + dH
    Next v
    '// 3) tab e maschera alla misura finale
    t.Width = m_F21_W("TabCtl_01") + dW
    t.Height = m_F21_H("TabCtl_01") + dH
    Me.Section(acDetail).Height = m_F21_H0 + dH
    Me.Width = m_F21_W0 + dW
    Me.Painting = True
    Exit Sub
ERRORE:
    Me.Painting = True
    Debug.Print Me.Name & " AdattaFinestra - errore " & Err.Number & ": " & Err.Description
End Sub
