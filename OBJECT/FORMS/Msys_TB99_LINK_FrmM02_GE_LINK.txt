'                               OGGETTI e PATH_DB affiancati; Pg_02..Pg_12: se la pagina ha UNA sola
'                               sottomaschera, questa occupa tutta la pagina.
'      #05.04# AdattaSottomaschera = se la form contenuta ha la procedura pubblica AdattaFinestra
'                               (es. UTILITA_Msys_Frm21_IMPORTA_OGGETTI) la richiama: i suoi controlli
'                               si allargano allo spazio della pagina.
'==================================================================================================

'//#05# misure (twip)
Private Const TAB_W_MIN As Long = 19332     ' misura di progetto del tab: sotto non si scende
Private Const TAB_H_MIN As Long = 8460
Private Const MARG As Long = 60             ' margine tra finestra e tab, e dentro le pagine
Private m_dxL As Long, m_dyT As Long, m_dxR As Long, m_dyB As Long
Private m_TabL As Long, m_TabT As Long
Private m_Pronto As Boolean

Private Sub Form_Load()
    '#01# misure di partenza, prima pagina e catena allineata
    On Error Resume Next
    MisureIniziali
    Me!TabCtl_01.Value = 0
    AggiornaPagina 0
End Sub

Private Sub Form_Resize()
    '#05# la finestra e' cambiata (apertura, ingrandimento, ridimensionamento)
    AdattaLayout
End Sub

Private Sub TabCtl_01_Change()
    '#02# cambio pagina
    AggiornaPagina Me!TabCtl_01.Value
End Sub

Private Sub AggiornaPagina(ByVal nPagina As Integer)
    '#03# aggiorna le sottomaschere della pagina selezionata (nPagina parte da 0 = Pg_01)
    Dim ctl As Control
    On Error Resume Next
    Select Case nPagina
        Case 0      ' Pg_01 - COLLEGAMENTI: catena delle 3 sottomaschere
            Me!SottoForm_LINK.Requery
            Me!TXT_NAME.Requery
            Me!SottoForm_OGGETTI.Requery
            Me!TXT_ID_OBJ.Requery
            Me!SottoForm_PATH_DB.Requery
            Me!SottoForm_LINK.SetFocus
        Case 1      ' Pg_02 - IMPORTA OGGETTI
            Me!SottoForm_IMPORTA.Requery
            Me!SottoForm_IMPORTA.SetFocus
        Case Else   ' Pg_03..Pg_12 - ogni sottomaschera presente nella pagina
            For Each ctl In Me!TabCtl_01.Pages(nPagina).Controls
                If ctl.ControlType = acSubform Then ctl.Requery
            Next ctl
    End Select
End Sub

Private Sub Cmd_HELP_Click()
    '#04# help HTML della linguetta attiva
    Select Case Me!TabCtl_01.Value
        Case 0:    HELP_Apri_pf "FORM", Me.Name, "01_COLLEGAMENTI"
        Case 1:    HELP_Apri_pf "FORM", Me.Name, "02_IMPORTA_OGGETTI"
        Case Else: HELP_Apri_pf "FORM", Me.Name
    End Select
End Sub

Private Sub MisureIniziali()
    '#05.01# bordi tra il tab e l'area utile delle pagine, letti dall'ultima pagina (vuota)
    Dim t As Control, p As Page
    On Error GoTo ERRORE
    Set t = Me!TabCtl_01
    Set p = t.Pages(t.Pages.Count - 1)
    m_TabL = t.Left: m_TabT = t.Top
    m_dxL = p.Left - t.Left
    m_dyT = p.Top - t.Top
    m_dxR = (t.Left + t.Width) - (p.Left + p.Width)
    m_dyB = (t.Top + t.Height) - (p.Top + p.Height)
    m_Pronto = True
    Exit Sub
ERRORE:
    m_Pronto = False
End Sub

Private Sub AdattaLayout()
    '#05.02# tab grande quanto la finestra (mai sotto la misura di progetto)
    Dim t As Control, wT As Long, hT As Long, wF As Long, hS As Long
    If Not m_Pronto Then MisureIniziali
    If Not m_Pronto Then Exit Sub
    On Error GoTo ERRORE
    Set t = Me!TabCtl_01
    wT = Me.InsideWidth - m_TabL - MARG
    hT = Me.InsideHeight - Me.Section(acHeader).Height - m_TabT - MARG
    If wT < TAB_W_MIN Then wT = TAB_W_MIN
    If hT < TAB_H_MIN Then hT = TAB_H_MIN
    If wT = t.Width And hT = t.Height Then Exit Sub
    wF = m_TabL + wT + MARG
    hS = m_TabT + hT + MARG
    Me.Painting = False
    '// 1) contenitori al massimo tra misura attuale e nuova
    If Me.Width < wF Then Me.Width = wF
    If Me.Section(acDetail).Height < hS Then Me.Section(acDetail).Height = hS
    t.Move m_TabL, m_TabT, IIf(wT > t.Width, wT, t.Width), IIf(hT > t.Height, hT, t.Height)
    '// 2) pagine e pulsante ? alla misura nuova
    DisponiPagine m_TabL + m_dxL, m_TabT + m_dyT, wT - m_dxL - m_dxR, hT - m_dyT - m_dyB
    Me!Cmd_HELP.Left = wF - Me!Cmd_HELP.Width - MARG
    '// 3) tab e maschera alla misura finale
    t.Move m_TabL, m_TabT, wT, hT
    Me.Section(acDetail).Height = hS
    Me.Width = wF
    Me.Painting = True
    Exit Sub
ERRORE:
    Me.Painting = True
    Debug.Print Me.Name & " AdattaLayout - errore " & Err.Number & ": " & Err.Description
End Sub

Private Sub DisponiPagine(ByVal pL As Long, ByVal pT As Long, ByVal pW As Long, ByVal pH As Long)
    '#05.03# disposizione dei controlli nell'area utile delle pagine (pL, pT, pW, pH)
    Const LH As Long = 300, GAP As Long = 90
    Dim x As Long, y As Long, iw As Long, avail As Long, hTop As Long, hBot As Long
    Dim half As Long, y2 As Long, x2 As Long, i As Integer, n As Integer
    Dim ctl As Control, sf As Control
    '// Pg_01: catena delle tre sottomaschere
    x = pL + MARG: y = pT + MARG: iw = pW - 2 * MARG
    avail = pH - 2 * MARG - 2 * (LH + 30) - GAP
    hTop = CLng(avail * 0.52): hBot = avail - hTop
    half = (iw - MARG) \ 2
    x2 = x + half + MARG
    y2 = y + LH + 30 + hTop + GAP
    Me!LBL_LINK.Move x, y
    Me!SottoForm_LINK.Move x, y + LH + 30, iw, hTop
    Me!LBL_OGGETTI.Move x, y2, half
    Me!SottoForm_OGGETTI.Move x, y2 + LH + 30, half, hBot
    Me!LBL_PATH_DB.Move x2, y2, iw - half - MARG
    Me!SottoForm_PATH_DB.Move x2, y2 + LH + 30, iw - half - MARG, hBot
    '// Pg_02..Pg_12: l'unica sottomaschera della pagina occupa tutta la pagina
    For i = 1 To Me!TabCtl_01.Pages.Count - 1
        n = 0
        For Each ctl In Me!TabCtl_01.Pages(i).Controls
            If ctl.ControlType = acSubform Then n = n + 1: Set sf = ctl
        Next ctl
        If n = 1 Then
            sf.Move pL, pT, pW, pH
            AdattaSottomaschera sf
        End If
    Next i
End Sub

Private Sub AdattaSottomaschera(ByVal sf As Control)
    '#05.04# la form contenuta si adatta allo spazio, se sa farlo (procedura pubblica AdattaFinestra)
    On Error Resume Next
    sf.Form.AdattaFinestra
End Sub
