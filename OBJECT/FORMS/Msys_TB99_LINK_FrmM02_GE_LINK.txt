'==================================================================================================
' Msys_TB99_LINK_FrmM02_GE_LINK  -  catalogo TB99 (maschera contenitore, non associata)
'   TabCtl_01 = 12 pagine, dimensionate sulla maschera UTILITA_Msys_Frm21_IMPORTA_OGGETTI
'     Pg_01 COLLEGAMENTI     : catena delle 3 sottomaschere
'        SottoForm_LINK     = Msys_TB99_LINK_FrmS05_LINK     (Msys_TB99_LINK)
'        SottoForm_OGGETTI  = Msys_TB99_LINK_FrmS03_OGGETTI  (Msys_TB99_LINK_Object,  legame TXT_NAME   -> Name)
'        SottoForm_PATH_DB  = Msys_TB99_LINK_FrmS04_PATH_DB  (Msys_TB99_LINK_PATH_DB, legame TXT_ID_OBJ -> ID_OBJ)
'     Pg_02 IMPORTA OGGETTI  : SottoForm_IMPORTA = UTILITA_Msys_Frm21_IMPORTA_OGGETTI (a misura, senza bordo e barre)
'     Pg_03..Pg_12           : nascoste (usi futuri)
'   Le linguette sono i pulsanti Cmd_Pg01 / Cmd_Pg02 nell'intestazione, che resta sempre visibile:
'   la maschera UTILITA e stata ridotta all'80% (17.416 x 7.845 twip) per stare senza barre di scorrimento.
' #01# Form_Load   = apre sulla prima pagina con la catena allineata
' #02# MostraPagina = attiva una pagina, evidenzia il suo pulsante e torna in alto a sinistra
' #03# Cmd_Pg01_Click / Cmd_Pg02_Click = pulsanti-linguetta
'==================================================================================================
Private Sub Form_Load()
    '#01# prima pagina e catena allineata
    On Error Resume Next
    MostraPagina 0
    Me!TXT_NAME.Requery
    Me!SottoForm_OGGETTI.Requery
    Me!TXT_ID_OBJ.Requery
    Me!SottoForm_PATH_DB.Requery
End Sub

Private Sub MostraPagina(ByVal nPagina As Integer)
    '#02# pagina attiva, pulsante in grassetto, fuoco sul primo controllo della pagina
    On Error Resume Next
    Me!TabCtl_01.Value = nPagina
    Me!Cmd_Pg01.FontBold = (nPagina = 0)
    Me!Cmd_Pg02.FontBold = (nPagina = 1)
    If nPagina = 0 Then Me!SottoForm_LINK.SetFocus Else Me!SottoForm_IMPORTA.SetFocus
End Sub

Private Sub Cmd_Pg01_Click()
    '#03# linguetta 1 - COLLEGAMENTI
    MostraPagina 0
End Sub

Private Sub Cmd_Pg02_Click()
    '#03# linguetta 2 - IMPORTA OGGETTI
    MostraPagina 1
End Sub
